cmake_minimum_required(VERSION 3.22)
option(TSL_USB_RELAY_BENCH "Build the isolated USB relay diagnostic, never deployment firmware" OFF)
if(TSL_USB_RELAY_BENCH)
    if(CMAKE_BINARY_DIR STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/build")
        message(FATAL_ERROR "Keep the production build intact; use -B build-bench for the relay diagnostic")
    endif()
    message(WARNING "USB RELAY BENCH IMAGE: isolated contacts only; restore production before deployment")
endif()
set(COMPONENTS main)
set(PROJECT_VER "0.2.0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(TSL_USB_RELAY_BENCH)
    project(tsl_relay_bench VERSION 0.2.0)
else()
    project(tsl_smart_contactor VERSION 0.2.0)
endif()
if(NOT "$ENV{IDF_VERSION}" STREQUAL "5.5.2")
    message(FATAL_ERROR "Use pinned ESP-IDF v5.5.2 (see tools/bootstrap.sh)")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "$ENV{IDF_PATH}" apply --reverse --check
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/patches/esp-idf-v5.5.2-https-oom.patch"
    RESULT_VARIABLE SDK_CLEANUP_PATCH_STATUS OUTPUT_QUIET ERROR_QUIET)
if(NOT SDK_CLEANUP_PATCH_STATUS EQUAL 0)
    message(FATAL_ERROR "Apply the pinned HTTPS cleanup fix: tools/bootstrap.sh --patch-sdk $ENV{IDF_PATH}")
endif()
